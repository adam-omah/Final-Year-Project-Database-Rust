Database Query Language Syntax Examples

SELECT Statements

            
                SELECT * FROM <table_name>
                SELECT <column1>, <column2>, <column3> FROM <table_name>
                SELECT * FROM <table_name> AT <timestamp>
                SELECT * FROM <table_name> WHERE <column> LIKE '<pattern>'
                SELECT * FROM <table_name> WHERE <column> = <value>
                SELECT <column1>, <column2> FROM <table_name> WHERE <column> >= <value>

CREATE TABLE Statements

            
                CREATE TABLE <table_name> (
                    <column1> <data_type>,
                    <column2> <data_type> UNIQUE,
                    <column3> <data_type> NOT NULL CHECK (<condition>)
                )

INSERT Statements

            
                INSERT INTO <table_name> (<column1>, <column2>) VALUES ("<value1>", <value2>)
                INSERT INTO <table_name> VALUES (<value1>, "<value2>")

UPDATE Statements

            
                UPDATE <table_name>
                SET <column> = <new_value>
                WHERE uuid = "<specific_uuid>"

DELETE Statements

            
                DELETE FROM <table_name>
                WHERE uuid = "<specific_uuid>"

Table Management

DROP TABLE <table_name>
